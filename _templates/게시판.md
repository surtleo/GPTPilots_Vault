<%*
const authorMap = {
  "beomjinkim2000": "김범진(PM)",
  "YOUR_GITHUB_ID_B": "팀원B(Exp)",
  "YOUR_GITHUB_ID_C": "팀원C(Model)",
  "YOUR_GITHUB_ID_D": "팀원D(Data)"
};

const configPath = "_config/me.md";
let gitUser = "";
try {
  const f = app.vault.getAbstractFileByPath(configPath);
  if (f) gitUser = (await app.vault.read(f)).split("\n")[0].trim();
} catch(e) {}

if (!gitUser) {
  gitUser = ((await tp.system.prompt("GitHub ID를 입력하세요 (한 번만 저장됩니다)")) || "").trim();
  // 빈 값이면 저장하지 않는다 — 빈 me.md가 생기면 이후 영영 다시 묻는 루프에 빠진다
  if (gitUser) {
    const meBody = gitUser + "\n\n> 게시판 글쓴이 자동 기록용 개인 설정 파일입니다 (1행 = 내 GitHub ID).\n> 이 파일을 지우면 새 글을 쓸 때마다 GitHub ID를 다시 물어봅니다.\n";
    try {
      if (!app.vault.getAbstractFileByPath("_config")) await app.vault.createFolder("_config");
      const existing = app.vault.getAbstractFileByPath(configPath);
      if (existing) await app.vault.modify(existing, meBody);
      else await app.vault.create(configPath, meBody);
    } catch(e) { console.error("[게시판 템플릿] me.md 저장 실패:", e); }
  }
}

const author = authorMap[gitUser] || gitUser || "익명";

const typeChoice = (await tp.system.suggester(
  ["💬 자유게시판", "❓ QnA게시판"],
  ["free", "qna"],
  false,
  "타입을 선택하세요"
)) || "free";

// 파일명에 못 쓰는 문자가 있으면 tp.file.move가 예외로 죽어 '제목 없음' 노트로 남는다 → 제거 후 이동
let title = ((await tp.system.prompt("제목")) || "").replace(/[\\/:*?"<>|]/g, "").trim();
if (!title) title = "새 글 " + tp.date.now("MM-DD HHmm");

const prefix = typeChoice === "qna" ? "[Q&A]-" : "[자유]-";
let target = "게시판/" + prefix + title;
if (app.vault.getAbstractFileByPath(target + ".md")) target += " " + tp.date.now("HHmmss");
await tp.file.move(target);
