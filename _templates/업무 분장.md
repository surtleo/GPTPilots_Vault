<%*
// 오늘 보고가 이미 있으면 중복 생성하지 않고 기존 보고를 연다
const dailyName = tp.date.now("YYYY-MM-DD") + "-" + tp.file.folder();
const existing = app.vault.getAbstractFileByPath(tp.file.folder(true) + "/" + dailyName + ".md");
if (existing) {
  new Notice("오늘 업무 보고가 이미 있어 기존 파일을 엽니다.");
  await app.workspace.getLeaf().openFile(existing);
  await app.vault.delete(tp.config.target_file);
  return;
}
await tp.file.rename(dailyName);
%>
